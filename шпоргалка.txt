async def create_something(session: SessionDep):
    # 1. Создаем объект (пока только в памяти Python)
    # ID еще нет.
    obj = MyModel(field="value")
    
    # 2. Добавляем в сессию (говорим "мы хотим это сохранить")
    # Объект попадает в "корзину покупок" SQLAlchemy.
    session.add(obj)
    
    # 3. Коммитим (отправляем в БД, физическое сохранение)
    # База данных присваивает ID и сохраняет строку.
    # Если здесь ошибка — ничего не сохранится.
    await session.commit()
    
    # 4. Обновляем (получаем созданный ID и дефолтные поля)
    # SQLAlchemy делает SELECT к базе, чтобы узнать новый ID.
    await session.refresh(obj)
    
    return obj
